10pub struct RatConvertQuery<M: ModInt> {
11    n: usize,
12    bin: Vec<(M, M)>,
13    prev: Vec<usize>,
14    next: Vec<usize>,
15}

17pub struct RatConvertRecipLogPowQuery<M: ModInt> {
18    rcq: RatConvertQuery<M>,
19    recip: Vec<M>,
20    log: Vec<M>,
21    giant_pow: Vec<M>,
22    baby_pow: Vec<M>,
23}

25impl<M: ModInt> RatConvertQuery<M> {
26    pub fn new(n: usize) -> Self {
27        const { assert!(M::IS_PRIME) };
28
29        let na = (M::new(0), M::new(0));
30        let mut bin = vec![na; n * n + 1];
31        let nn = (n * n) as u128;
32        for num in (1..=n).rev() {
33            for den in num..=n {
34                let i = (nn * num as u128 / den as u128) as usize;
35                bin[i] = (M::new(num), M::new(den));
36            }
37        }
38        bin[0] = (M::new(0), M::new(1));
39
40        let npos = usize::MAX;
41        let mut prev = vec![npos; n * n + 1];
42        let mut last = 0;
43        for i in 1..=n * n {
44            prev[i] = last;
45            if bin[i] != na {
46                last = i;
47            }
48        }
49        let mut next = vec![npos; n * n + 1];
50        let mut last = n * n;
51        for i in (0..n * n).rev() {
52            next[i] = last;
53            if bin[i] != na {
54                last = i;
55            }
56        }
57
58        Self { n, bin, prev, next }
59    }
60
61    pub fn rat_convert(&self, x: M) -> ((bool, M), M) {
62        let x_u32 = x.get();
63        if x_u32 == 0 || x_u32 == 1 {
64            return ((false, x), M::new(1));
65        }
66
67        let n = self.n;
68        let nn = (n * n) as u128;
69        let p = M::MOD;
70        let i = (nn * x_u32 as u128 / p as u128) as usize;
71
72        let (c, b) = self.bin[i];
73        let (c_u32, b_u32) = (c.get(), b.get());
74        let ((c1, b1), (c2, b2)) = if (c_u32, b_u32) == (0, 0) {
75            (self.bin[self.prev[i]], self.bin[self.next[i]])
76        } else if c_u32 as u64 * p as u64 <= b_u32 as u64 * x_u32 as u64 {
77            // c/b <= x/p
78            ((c, b), self.bin[self.next[i]])
79        } else {
80            (self.bin[self.prev[i]], (c, b))
81        };
82        let (b, c) = if b1.get() < b2.get() { (b1, c1) } else { (b2, c2) };
83        let (b_u32, c_u32) = (b.get(), c.get());
84        let a = b * x;
85        let a_is_neg =
86            (b_u32 as u64 * x_u32 as u64) < (c_u32 as u64 * p as u64);
87        ((a_is_neg, a), b)
88    }
89}

91fn ceil_cbrt(n: usize) -> usize {
92    (0_usize..).find(|&i| i.pow(3) >= n).unwrap()
93}

95impl<M: ModInt> RatConvertRecipLogPowQuery<M> {
96    pub fn new(r: M) -> Self {
97        let n = ceil_cbrt(M::MOD as usize);
98        // r must be a primitive root of M::MOD
99
100        let rcq = RatConvertQuery::new(n);
101        let recip = M::recip_table(2 * M::MOD as usize / n);
102        let log = Self::log_table(2 * M::MOD as usize / n, r);
103
104        let giant_pow = r.pow(B).pow_table(B); // FIXME
105        let baby_pow = r.pow_table(B); // FIXME
106
107        Self { rcq, recip, log, giant_pow, baby_pow }
108    }
109
110    fn log_table(k: usize, r: M) -> Vec<M> {
111        let ls = LinearSieve::new(k);
112        let b = 200000; // FIXME
113        let bsgs = Bsgs::<M>::new(b, r);
114        let p_usize = M::MOD as usize;
115        let log_neg1 = M::new((p_usize - 1) / 2);
116        let mut res = vec![M::new(0); k + 1];
117        for i in 2..=k {
118            let lpf_i = ls.lpf(i).unwrap();
119            res[i] = if lpf_i < i {
120                res[lpf_i] + res[i / lpf_i]
121            } else if i * i >= p_usize {
122                let j = p_usize / i;
123                let k = p_usize % i;
124                log_neg1 + res[k] - res[j]
125            } else {
126                M::new(bsgs.log(M::new(i)))
127            };
128        }
129        res
130    }
131
132    pub fn rat_convert(&self, x: M) -> ((bool, M), M) {
133        self.rcq.rat_convert(x)
134    }
135
136    pub fn recip(&self, x: M) -> M {
137        assert_ne!(x, M::new(0));
138
139        let ((a_is_neg, a), b) = self.rat_convert(x);
140        if a_is_neg {
141            -self.recip[(-a).get() as usize] * b
142        } else {
143            self.recip[a.get() as usize] * b
144        }
145    }
146
147    pub fn log(&self, x: M) -> M {
148        let ((a_is_neg, a), b) = self.rat_convert(x);
149        let log_neg1 = M::new((M::MOD as u32 - 1) / 2);
150        let log_b = self.log[b.get() as usize];
151
152        if a_is_neg {
153            // log(-|a|/b) = log(-1) + log(|a|) - log(b)
154            log_neg1 + self.log[(-a).get() as usize] - log_b
155        } else {
156            self.log[a.get() as usize] - log_b
157        }
158    }
159
160    pub fn pow(&self, a: M, b: usize) -> M {
161        if a.get() == 0 {
162            return if b == 0 { M::new(1) } else { a };
163        }
164
165        let log_a = self.log(a);
166        let e =
167            ((b as u64 * log_a.get() as u64) % (M::MOD - 1) as u64) as usize;
168        self.giant_pow[e / B] * self.baby_pow[e % B]
169    }
170}

172#[test]
173fn sanity_check() {
174    const P: u32 = 1000003;
175    type Mi = modint::StaticModInt<P>;
176
177    let n = 101;
178    let rcq = RatConvertQuery::<Mi>::new(n);
179
180    for x_u32 in 0..P {
181        let x = Mi::new(x_u32);
182        let ((a_is_neg, a), b) = rcq.rat_convert(x);
183        assert_eq!(a / b, x);
184
185        let a_abs_u32 = if a_is_neg { P - a.get() } else { a.get() };
186        let b_u32 = b.get();
187        assert!(a_abs_u32 <= 2 * P / n as u32);
188        assert!(1 <= b_u32 && b_u32 <= n as u32);
189    }
190}

192#[test]
193fn recip() {
194    type Mi = modint::ModInt998244353;
195    const P: u32 = Mi::MOD;
196
197    let r = Mi::new(3);
198    let rcq = RatConvertRecipLogPowQuery::<Mi>::new(r);
199
200    for x_u32 in (1..P).step_by(20) {
201        let x = Mi::new(x_u32);
202
203        let x_recip = rcq.recip(x);
204        assert_eq!(x * x_recip, Mi::new(1));
