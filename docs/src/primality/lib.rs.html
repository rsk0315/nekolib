1macro_rules! impl_is_sprp {
2    ( $( ($fn_name:ident, $word:ty, $dword:ty) ),* ) => { $(
3        pub const fn $fn_name(n: $word, a: $word) -> bool {
4            let s = (n - 1).trailing_zeros();
5            let d = n >> s;
6            let mut cur = {
7                let mut cur = 1;
8                let mut pow = d;
9                let mut a = a;
10                while pow > 0 {
11                    if pow & 1 != 0 {
12                        cur = (cur as $dword * a as $dword % n as $dword) as $word;
13                    }
14                    a = ((a as $dword).pow(2) % n as $dword) as $word;
15                    pow >>= 1;
16                }
17                cur
18            };
19            if cur == 1 {
20                return true;
21            }
22            let mut i = 0;
23            while i < s {
24                if cur == n - 1 {
25                    return true;
26                }
27                cur = ((cur as $dword).pow(2) % n as $dword) as $word;
28                i += 1;
29            }
30            false
31        }
32    )* }
33}

73pub const fn is_prime_u8(n: u8) -> bool {
74    if n == 2 || n == 3 || n == 5 || n == 7 || n == 11 || n == 13 {
75        return true;
76    }
77    n > 1
78        && n % 2 > 0
79        && n % 3 > 0
80        && n % 5 > 0
81        && n % 7 > 0
82        && n % 11 > 0
83        && n % 13 > 0
84}

88pub const fn is_prime_u32(n: u32) -> bool {
89    if n == 2 || n == 3 || n == 5 || n == 7 {
90        return true;
91    }
92    if n % 2 == 0 || n % 3 == 0 || n % 5 == 0 || n % 7 == 0 {
93        return false;
94    }
95    if n < 121 {
96        return n > 1;
97    }
98    let h = n as u64;
99    let h = ((h >> 16) ^ h).wrapping_mul(0x45D9F3B);
100    let h = ((h >> 16) ^ h).wrapping_mul(0x45D9F3B);
101    let h = ((h >> 16) ^ h) & 0xFF;
102    is_sprp_32(n, BASES[h as usize] as u32)
103}

105pub const fn is_prime_u64(n: u64) -> bool {
106    if n == 2 || n == 3 || n == 5 || n == 7 {
107        return true;
108    }
109    if n % 2 == 0 || n % 3 == 0 || n % 5 == 0 || n % 7 == 0 {
110        return false;
111    }
112    if n < 121 {
113        return n > 1;
114    }
115    let bases = [2, 325, 9375, 28178, 450775, 9780504, 1795265022];
116    let mut i = 0;
117    while i < bases.len() {
118        let b = bases[i];
119        if !(b % n == 0 || is_sprp_64(n, b % n)) {
120            return false;
121        }
122        i += 1;
123    }
124    true
125}

127#[test]
128fn exhaustive_u32() {
129    let w = 64;
130    let n = 1_usize << 32;
131    let is_prime = {
132        let mut dp = vec![!0_u64; n / w + 1];
133        dp[0] &= !0 << 2;
134        for i in (2..=n).take_while(|&i| i <= n / i) {
135            let (qi, ri) = (i / w, i % w);
136            if dp[qi] >> ri & 1 == 0 {
137                continue;
138            }
139            for j in i..=n / i {
140                let (qj, rj) = (i * j / w, i * j % w);
141                dp[qj] &= !(1 << rj);
142            }
143        }
144        dp
145    };
146
147    for i in 2..n {
148        let actual = is_prime_u32(i as u32);
149        let expected = is_prime[i / w] >> (i % w) & 1 != 0;
150        assert_eq!(actual, expected);
151    }
