1use modint::ModInt;
2
3#[derive(Clone, Debug)]
4pub struct FactorialTable<M> {
5    fact: Vec<M>,
6    fact_recip: Vec<M>,
7}
8
9impl<M: ModInt> FactorialTable<M> {
10    pub fn new(n: usize) -> Self {
11        let mut fact = vec![M::new(1); n + 1];
12        for i in 0..n {
13            fact[i + 1] = fact[i] * M::new(i + 1);
14        }
15        let recip = M::recip_table(n);
16        let mut fact_recip = vec![M::new(1); n + 1];
17        for i in 0..n {
18            fact_recip[i + 1] = fact_recip[i] * recip[i + 1];
19        }
20        Self { fact, fact_recip }
21    }
22
23    pub fn factorial(&self, i: usize) -> M { self.fact[i] }
24    pub fn factorial_recip(&self, i: usize) -> M { self.fact_recip[i] }
25
26    pub fn perm(&self, i: usize, j: usize) -> M {
27        if j <= i { self.fact[i] * self.fact_recip[i - j] } else { M::new(0) }
28    }
29    pub fn perm_recip(&self, i: usize, j: usize) -> M {
30        self.fact_recip[i] * self.fact[i - j]
31    }
32
33    pub fn binom(&self, i: usize, j: usize) -> M {
34        self.perm(i, j) * self.fact_recip[j]
35    }
36    pub fn binom_recip(&self, i: usize, j: usize) -> M {
37        self.perm_recip(i, j) * self.fact[j]
38    }
39
40    pub fn recip(&self, i: usize) -> M { self.fact_recip[i] * self.fact[i - 1] }
41}

43#[test]
44fn sanity_check() {
45    type Mi = modint::ModInt998244353;
46
47    let ft = FactorialTable::<Mi>::new(10);
48    assert_eq!(ft.factorial(7), Mi::new(5040));
49    assert_eq!(ft.factorial_recip(7), Mi::new(5040).recip());
50    assert_eq!(ft.binom(5, 2), Mi::new(10));
51    assert_eq!(ft.binom_recip(5, 2), Mi::new(10).recip());
52    assert_eq!(ft.recip(10), Mi::new(10).recip());
53}
