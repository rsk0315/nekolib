1//! 母関数を考える際に必要となる操作。
2//!
3//! ## Notations
4//!
5//! $`n`$ 次多項式 $`f(x) = \sum_{i=0}^n a_ix^i`$ に対し、$`[x^i]\, f(x) = a_i`$ とする。
6//! $`\gdef\e{\mathrm e}`$
7//! $`\gdef\d{\mathrm d}`$
8//!
9//! ## Basics
10//!
11//! $`a(x) = \sum_{i=0}^n a_ix^i`$ かつ $`b(x) = \sum_{i=0}^n b_ix^i`$ とする。
12//! ```math
13//! \begin{aligned}
14//! [x^i]\, \frac{a(x)}{1-x} &= \sum_{j=0}^i a_j, \\
15//! [x^i]\, \bigl(a(x)\cdot b(x)\bigr) &= \sum_{j=0}^i a_jb_{i-j}
16//! \end{aligned}
17//! ```
18//!
19//! すなわち、累積和や畳み込みは多項式の演算に対応すると見なせる。
20//!
21//! $`\sum`$ の添字の入れ替えを悩まずできると吉。
22//!
23//! ```math
24//! \begin{aligned}
25//! \sum_{i=L}^R f(i) &= \sum_{i=L+1}^{R+1} f(i-1)
26//! = f((L+1)-1) + f((L+2)-1) + \dots + f((R+1)-1), \\
27//! \sum_{i=L}^R f(i) &= \sum_{R-i=0}^{R-L} f(L+(R-i))
28//! = f(L+0) + f(L+1) + \dots + f(L+(R-L)) \\
29//! &= \sum_{R-i=0}^{R-L} f(R-(R-i))
30//! = f(R-0) + f(R-1) + \dots + f(R-(R-L)).
31//! \end{aligned}
32//! ```
33//!
34//! ## Convolution-like
35//!
36//! $`a_i = a^{\leftarrow}_{n-i}`$ とする。
37//!
38//! ```math
39//! \begin{aligned}
40//! \sum_{i=0}^n \left(\sum_{j=i}^n a_jb_{j-i}\right)x^i
41//! &= \sum_{i=0}^n \left(\sum_{j=i}^n a^{\leftarrow}_{n-j}b_{j-i}\right)x^i \\
42//! &= \sum_{i=0}^n \left(\sum_{n-j=0}^{n-i} a^{\leftarrow}_{n-(i+n-j)}b_{(i+n-j)-i}) \right) x^i \\
43//! &= \sum_{i=0}^n \left(\sum_{n-j=0}^{n-i} a^{\leftarrow}_{(n-i)-(n-j)}b_{n-j}) \right) x^i \\
44//! &= \sum_{n-i=0}^n \left(\sum_{n-j=0}^{n-i} a^{\leftarrow}_{(n-i)-(n-j)}b_{n-j}) \right) x^i \\
45//! &= \sum_{i'=0}^n \left(\sum_{j'=0}^{i'} a^{\leftarrow}_{i'-j'}b_{j'} \right) x^{n-i'} \\
46//! &= \sum_{i=0}^n \left([x^i]\, \bigl(a^{\leftarrow}(x)\cdot b(x)\bigr) \right) x^{n-i}. \\
47//! \end{aligned}
48//! ```

50//! ## Operations
51//!
52//! ### Reciprocal
53//!
54//! $`f(x)\cdot g(x) \bmod x^k = 1`$ なる $`g(x)`$ を求める。
55//!
56//! ### Square root
57//!
58//! $`g(x)^2 \bmod x^k = f(x)`$ なる $`g(x)`$ を求める。
59//!
60//! ### Exponential
61//!
62//! 下記で定義される $`\exp(f(x))`$ に対して、$`\exp(f(x))\bmod x^k`$ を求める。
63//! ```math
64//! \exp(f(x)) = \sum_{n=0}^{\infty} \frac{f(x)^n}{n!}.
65//! ```
66//!
67//! ### Logarithm
68//!
69//! $`[x^0]\, f(x) = 1`$ なる $`f`$ に対し、$`\log(f(x)) \bmod x^k`$ を求める。
70//! ```math
71//! \log(1-f(x)) = -\sum_{n=1}^{\infty} \frac{f(x)^n}n.
72//! ```
73//!
74//! $`\tfrac{\d}{\d x} \log(f(x)) = f'(x)\cdot f(x)^{-1}`$ や
75//! $`\log(f(x)\cdot g(x)) = \log(f(x)) + \log(g(x))`$ などが成り立つ。
76//!
77//! ### Power
78//!
79//! $`f(x)`$ と $`k`$ に対して $`f(x)^k`$ を求める。
80//! $`f(x)^0 = 1`$ および、$`k\ne 0`$ に対して $`0^k = 0`$ が成り立つ。
81//!
82//! 以下、$`f(x)\ne 0`$ かつ $`k\ne 0`$ とする。
83//! ある $`l`$ と $`g(x)`$ に対して $`f(x) = a_l x^l\cdot(1+g(x)\cdot x)`$ が成り立つので、
84//! ```math
85//! \begin{aligned}
86//! f(x)^k
87//! &= \bigl(a_lx^l\cdot (1+g(x)\cdot x)\bigr)^k \\
88//! &= a_l^k x^{lk} \cdot (1+g(x)\cdot x)^k \\
89//! &= a_l^k x^{lk} \cdot \exp(k\cdot \log(1+g(x)\cdot x))
90//! \end{aligned}
91//! ```
92//! により求められる。
93//!
94//! ### Circular
95//!
96//! $`\cos(f(x))`$ および $`\sin(f(x))`$ を求める。
97//!
98//! ### Polynomial equation
99//!
100//! $`\varphi(y) \equiv 0 \pmod{x^k}`$ なる $`y`$ を求める。
101//! $`(y, k) \mapsto \varphi(y)\cdot \left(\tfrac{\d}{\d y}\varphi(y)\right)^{-1} \bmod x^k`$ の
102//! oracle を用いる。$`\tfrac{\d x}{\d y} = 0`$ として考えるっぽい。
103//!
104//! ### First-order derivative equation
105//!
106//! $`y' \equiv f(y, x) \pmod{x^k}`$ を満たす $`y(x)`$ を求める。
107//! $`(y, k) \mapsto (f(y, x) \bmod x^k, f'(y, x) \bmod x^k)`$ の oracle を用いる。
108//!
109//! ### Relaxed multiplication
110//!
111//! $`f(x)`$ と $`g(x)`$ に対し、$`(f\cdot g)(x) = f(x)\cdot g(x)`$ を求める。ただし、$`[x^i]\, f(x)`$
112//! や $`[x^i]\, g(x)`$ は、各 $`0\le j\lt i`$ における $`[x^j]\, (f\cdot g)(x)`$
113//! を用いて計算されるとする。
114//!
115//! たとえば、与えられた $`\varphi`$ に対して $`\e^{\varphi}`$ を求めるのは、下記の微分方程式を解くことに帰着できる。
116//! ```math
117//! \e^{\varphi} = \int \varphi' \e^{\varphi}.
118//! ```
119//! $`f = \varphi'`$ かつ $`g = \e^{\varphi}`$ として relaxed multiplication を行い、 $`i\gt 0`$
120//! に対して $`[x^i]\, g(x) = i^{-1}\cdot [x^{i-1}]\, (f\cdot g)(x)`$ とすればよい。
121//!
122//! $`\mathrm C_n \mathrm H_{2n+1} \mathrm{OH}`$ の形で表されるアルコールの立体異性体の個数は
123//! ```math
124//! s(z) = 1 + z\cdot \frac{s(z)^3 + 2s(z^3)}3
125//! ```
126//! として $`[z^n]\, s(z)`$ として表せることが Pólya によって示されている。$`[z^0]\, s(z)^3 = 1`$
127//! であり、$`i\gt 0`$ に対して
128//! ```math
129//! \begin{aligned}
130//! [z^i]\, s(z) &= [z^{i-1}] \left(3^{-1}\cdot s(z)^3 + 2\cdot 3^{-1}\cdot s(z^3)\right) \\
131//! &= 3^{-1}\cdot [z^{i-1}]\, s(z)^3 + 2\cdot 3^{-1}\cdot [z^{i-1}]\, s(z^3) \\
132//! &= 3^{-1}\cdot [z^{i-1}]\, s(z)^3 + 2\cdot 3^{-1}\cdot [z^{(i-1)/3}]\, s(z) \\
133//! \end{aligned}
134//! ```
135//! なので…？
136//!
137//! ```math
138//! f(z) = z\cdot\left(1+f{\left(\frac z{1+z}\right)} - z^4 f'(z)^2\right)
139//! ```
140//! が $`O(n\log(n)^3\log(\log(n)))`$ 時間で解けるらしい。
141//! ```math
142//! f(z) = z + f(z^2+z^3)
143//! ```
144//! が $`\tilde O(n)`$ 時間？で解けるらしい。
145//! ```math
146//! f(z) = z + f(z f(z) + z^2 f'(z)) + z^4\exp(z f''(z))
147//! ```
148//! が $`O(n^{3/2}\log(n)^{5/2}\log(\log(n)))`$ 時間で解けるらしい。

164//! ### Log-exp
165//!
166//! $`\log(1-f(x))`$ は下記として定義される（再掲）。
167//! ```math
168//! \log(1-f(x)) = -\sum_{n=1}^{\infty} \frac{f(x)^n}n.
169//! ```
170//!
171//! $`\prod\varphi = \exp(\log(\prod\varphi)) = \exp(\sum \log(\varphi))`$
172//! を用いて式変形できる場合がある。
173//!
174//! たとえば次のような例がある。
175//! ```math
176//! \begin{aligned}
177//! &\phantom{{}={}} \prod_{i=1}^{n} {(1+x^i+x^{2i}+\dots+x^{i\cdot a_i})} \\
178//! &= \prod_{i=1}^n {\left(\frac1{1-x^i}-\frac{x^{i\cdot(a_i+1)}}{1-x^i}\right)} \\
179//! &= \prod_{i=1}^n {\frac{1-x^{i\cdot(a_i+1)}}{1-x^i}} \\
180//! &= {\exp}{\left({\log}{\left(\prod_{i=1}^n {\frac{1-x^{i\cdot(a_i+1)}}{1-x^i}}\right)}\right)} \\
181//! &= {\exp}{\left(\sum_{i=1}^n {{\log}{\left(\frac{1-x^{i\cdot(a_i+1)}}{1-x^i}\right)}}\right)} \\
182//! &= {\exp}{\left(\sum_{i=1}^n {{\log}{\left(\frac{1-x^{i\cdot(a_i+1)}}{1-x^i}\right)}}\right)} \\
183//! &= {\exp}{\left(\sum_{i=1}^n {{\log}{\left(1-x^{i\cdot(a_i+1)}\right)} - \sum_{i=1}^n {{\log}{\left({1-x^i}\right)}}}\right)} \\
184//! &= {\exp}{\left(\sum_{i=1}^n {\left(-\sum_{j=1}^{\infty} \frac{x^{i\cdot(a_i+1)\cdot j}}j\right)} - \sum_{i=1}^n {\left(-\sum_{j=1}^{\infty} \frac{x^{i\cdot j}}j\right)}\right)} \\
185//! &= {\exp}{\left(-\sum_{i=1}^n {\left(\sum_{j=1}^{\infty} \frac{x^{i\cdot(a_i+1)\cdot j}}j\right)} + \sum_{i=1}^n {\left(\sum_{j=1}^{\infty} \frac{x^{i\cdot j}}j\right)}\right)} \\
186//! &= {\exp}{\left(\sum_{i=1}^n {\left(\sum_{j=1}^{\infty} \frac{x^{i\cdot j}}j\right)} -\sum_{i=1}^n {\left(\sum_{j=1}^{\infty} \frac{x^{i\cdot(a_i+1)\cdot j}}j\right)}\right)}, \\
187//! &\phantom{{}={}} [x^n]\, {\exp}{\left(\sum_{i=1}^n {\left(\sum_{j=1}^{\infty} \frac{x^{i\cdot j}}j\right)} -\sum_{i=1}^n {\left(\sum_{j=1}^{\infty} \frac{x^{i\cdot(a_i+1)\cdot j}}j\right)}\right)} \\
188//! &= [x^n]\, {\exp}{\left(\sum_{i=1}^n {\left(\sum_{j=1}^{\floor{n/i}} \frac{x^{i\cdot j}}j\right)} -\sum_{i=1}^n {\left(\sum_{j=1}^{\floor{n/i}} \frac{x^{i\cdot(a_i+1)\cdot j}}j\right)}\right)}.
189//! \end{aligned}
190//! ```
191//! 調和級数に関する計算量解析により、$`\exp`$ の引数は $`O(n\log(n))`$ 時間で計算できる。See
192//! also [FPS-24 N](https://atcoder.jp/contests/fps-24/tasks/fps_24_n).
193//!
194//! ### Sum of powers
195//!
196//! 与えられた $`a = (a_0, a_1, \dots, a_{m-1})`$ に対し、各 $`i\ge 0`$ について
197//! $`\sum_{j=0}^{m-1} a_j^i`$ を求めたいとする。
198//!
199//! ```math
200//! \begin{aligned}
201//! &\phantom{{}={}} -\frac{\d}{\d x}{\log}{\left(\prod_{j=0}^{m-1} {(1-a_j x)}\right)} \\
202//! &= -\frac{\d}{\d x} \sum_{j=0}^{m-1} {\log(1-a_j x)} \\
203//! &= -\sum_{j=0}^{m-1} {\frac{\d}{\d x} \log(1-a_j x)} \\
204//! &= -\sum_{j=0}^{m-1} {\frac{(1-a_j x)'}{1-a_j x}} \\
205//! &= -\sum_{j=0}^{m-1} {-\frac{a_j}{1-a_j x}} \\
206//! &= \sum_{j=0}^{m-1} {\frac{a_j}{1-a_j x}} \\
207//! &= \sum_{j=0}^{m-1} \sum_{i=0}^{\infty} a_j^{i+1} x^i \\
208//! &= \sum_{i=0}^{\infty} \sum_{j=0}^{m-1} a_j^{i+1} x^i \\
209//! \end{aligned}
210//! ```
211//! より、
212//! ```math
213//! [x^n]\, \left(-\frac{\d}{\d x}{\log}{\left(\prod_{j=0}^{m-1} {(1-a_j x)}\right)}\right)
214//! = \sum_{j=0}^{m-1} a_j^{i+1}
215//! ```
216//! や
217//! ```math
218//! \sum_{i=0}^{\infty} \sum_{j=0}^{m-1} a_j^i x^i
219//! = m - x \cdot \left(\frac{\d}{\d x}{\log}{\left(\prod_{j=0}^{m-1} {(1-a_j x)}\right)}\right)
220//! ```
221//! が成り立つ。
222//!
223//! あるいは、
224//! ```math
225//! \frac{P_1(x)}{Q_1(x)} + \frac{P_2(x)}{Q_2(x)}
226//! = \frac{(P_1\cdot Q_2 + P_2\cdot Q_1)(x)}{(Q_1\cdot Q_2)(x)}
227//! ```
228//! より、各 $`i`$ に対して $`P_i(x) = 1`$ かつ $`Q_i(x) = 1-a_i x`$
229//! として（適切に分割統治を用いて）総和を求めることで、対応する有理式を $`O(n\log(n)^2)`$
230//! 時間で求めることもできる。
