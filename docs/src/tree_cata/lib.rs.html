1use std::collections::VecDeque;
2
3pub struct TreeCata<T> {
4    par: Vec<Option<(usize, usize, T)>>,
5    order: Vec<(usize, usize)>,
6    child: Vec<Vec<(usize, usize, T)>>,
7}
8
9impl<T> From<Vec<Vec<(usize, T)>>> for TreeCata<T> {
10    fn from(mut g: Vec<Vec<(usize, T)>>) -> Self {
11        let n = g.len();
12        let mut par: Vec<_> = (0..n).map(|_| None).collect();
13        let mut q: VecDeque<_> = vec![(0, n)].into();
14        let mut order = vec![];
15        let mut child: Vec<_> = (0..n).map(|_| vec![]).collect();
16
17        while let Some((v, vi)) = q.pop_front() {
18            order.push((v, vi));
19            let gv = std::mem::take(&mut g[v]);
20            for (i, (nv, w)) in (0..).zip(gv) {
21                if nv == 0 || par[nv].is_some() {
22                    par[v] = Some((nv, i, w));
23                } else {
24                    child[v].push((nv, i, w));
25                    q.push_back((nv, i));
26                }
27            }
28        }
29
30        Self { par, order, child }
31    }
32}

34impl<T> TreeCata<T> {
35    pub fn cata<U: Clone>(
36        &self,
37        empty: U,
38        mut map: impl FnMut(&U, &T) -> U,
39        mut fold: impl FnMut(&U, &U) -> U,
40    ) -> (Vec<U>, Vec<Vec<U>>) {
41        let n = self.child.len();
42        if n == 0 {
43            return (vec![], vec![]);
44        }
45
46        // dp_fold[v] := 0 を根としたときの v 以下の部分木の fold
47        // dp_map[v][i] := v を根としたときの i 番目の map
48        // accl[v][i] := v を根としたときの ..i 番目の子の fold
49        // accr[v][i] := v を根としたときの i.. 番目の子の fold
50
51        let len = |v: usize| self.child[v].len() + if v == 0 { 0 } else { 1 };
52
53        let empty = || empty.clone();
54        let mut dp_fold = vec![empty(); n];
55        let mut dp_map: Vec<_> =
56            (0..n).map(|v| vec![empty(); len(v)]).collect();
57        for &(v, vi) in self.order[1..].iter().rev() {
58            dp_fold[v] = dp_map[v].iter().fold(empty(), |x, y| fold(&x, y));
59            if let Some((pv, _, ref w)) = self.par[v] {
60                dp_map[pv][vi] = map(&dp_fold[v], &w);
61            }
62        }
63
64        let mut accl: Vec<_> =
65            (0..n).map(|v| vec![empty(); len(v) + 1]).collect();
66        let mut accr = accl.clone();
67
68        for &(v, _) in &self.order {
69            let children = &self.child[v];
70            for i in 0..len(v) {
71                accl[v][i + 1] = fold(&accl[v][i], &dp_map[v][i]);
72            }
73            for i in (0..len(v)).rev() {
74                accr[v][i] = fold(&dp_map[v][i], &accr[v][i + 1]);
75            }
76
77            for &(u, i, ref w) in children {
78                let j = self.par[u].as_ref().unwrap().1;
79                let tmp = fold(&accl[v][i], &accr[v][i + 1]);
80                dp_map[u][j] = map(&tmp, w);
81            }
82        }
83
84        // dp_fold[v] := v を根としたときの部分木の fold
85        let mut dp_fold = vec![empty(); n];
86        for v in 0..n {
87            dp_fold[v] = accr[v][0].clone();
88        }
89
90        (dp_fold, dp_map)
91    }
92}
